QUESTION: Write a program in C++ to swap two numbers using pointers.
CODE:
#include <iostream>
using namespace std;
void swapNum(int *x, int *y)
{
    int t = *x;
    *x = *y;
    *y = t;
}
int main()
{
    int a, b;
    cout << "Enter two numbers: ";
    cin >> a >> b;
    cout << "Before swap: a = " << a << " b = " << b << endl;
    swapNum(&a, &b);
    cout << "After swap: a = " << a << " b = " << b << endl;
    return 0;
}
OUTPUT:
Enter two numbers: 10 20
Before swap: a = 10 b = 20
After swap: a = 20 b = 10

CODE:
#include <iostream>
using namespace std;
int main()
{
    int marks;
    cout << "Enter marks: ";
    cin >> marks;
    try
    {
        if (marks < 0 || marks > 100)
        {
            throw marks;
        }
        if (marks >= 40)
        {
            cout << "Student Passed" << endl;
        }
        else
        {
            cout << "Student Failed" << endl;
        }
    }
    catch (int m)
    {
        cout << "Invalid Marks: " << m << endl;
    }
    return 0;
}

QUESTION: Write a C++ program to create a class Bank with functions to open account, deposit, withdraw, and show balance.
CODE:
#include <iostream>
using namespace std;
class Bank
{
public:
    int acno;
    string name;
    float bal;
    void openAcc(int a, string nm, float ob)
    {
        acno = a;
        name = nm;
        bal = ob;
        cout << "Account opened for " << name << endl;
    }
    void deposit(float amt)
    {
        bal = bal + amt;
        cout << name << ", Rs." << amt << " deposited." << endl;
    }
    void withdraw(float amt)
    {
        if (bal >= amt)
        {
            bal = bal - amt;
            cout << name << ", Rs." << amt << " debited." << endl;
        }
        else
        {
            cout << name << ", Insufficient balance" << endl;
        }
    }
    void showBal()
    {
        cout << "Name: " << name << "\tBal: " << bal << endl;
    }
};
int main()
{
    Bank a1, a2;
    a1.openAcc(1001, "Hardik", 5000);
    a2.openAcc(1002, "Pandya", 600);
    a1.deposit(3000);
    a1.deposit(4000);
    a2.deposit(1000);
    a2.withdraw(50000);
    a1.withdraw(2222);
    a1.showBal();
    a2.showBal();
    return 0;
}

Account opened for Pandya

Hardik, Rs.4000 deposited.
